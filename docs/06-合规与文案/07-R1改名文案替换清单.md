  Where-Object { $_.FullName -notmatch '\\design\\' -and $_.FullName -notmatch '\\node_modules\\' -and

                 $_.FullName -notmatch '\\reports\\' } |
  Select-String -Pattern '求职避雷' -Encoding UTF8
